#include <stdbool.h>
#include <stdint.h>

#include "inc/hw_memmap.h"
#include "driverlib/gpio.h"
#include "driverlib/sysctl.h"

#include "HAL/HAL_Buttons.h"

void HAL_Button_Init(void)
{
    SysCtlPeripheralEnable(SYSCTL_PERIPH_GPIOJ);
    while (!SysCtlPeripheralReady(SYSCTL_PERIPH_GPIOJ))
    {
    }

    GPIOPinTypeGPIOInput(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2));
    GPIOPadConfigSet(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2),
                     GPIO_STRENGTH_2MA, GPIO_PIN_TYPE_STD_WPU);
}

bool buttonState(Button_t button)
{
    /* Boutons actifs bas : relâché = 1 (pull-up), appuyé = 0. */
    return (GPIOPinRead(GPIO_PORTJ_BASE, (uint8_t)button) == 0);
}

/* main.c — exercice 1.2 : les LEDs suivent les boutons, sur interruption. */
#include <stdbool.h>
#include <stdint.h>

#include "HAL/HAL_TM4C1294.h"
#include "HAL/HAL_Buttons.h"
#include "HAL/HAL_Leds.h"

int main(void)
{
    clockInit();
    HAL_Button_Init();      /* broches + interruption */
    HAL_Led_Init();

    while (1)
    {
        switch (HAL_Button_Event(eBUTTON_USR_SW1))
        {

            case eBUTTON_EVENT_RELEASED: HAL_Led_Write(eLED1, false); break;
            default: break;
        }

        switch (HAL_Button_Event(eBUTTON_USR_SW2))
        {
            case eBUTTON_EVENT_PRESSED:  HAL_Led_Write(eLED2, true);  break;
            case eBUTTON_EVENT_RELEASED: HAL_Led_Write(eLED2, false); break;
            default: break;
        }
    }
}

/* HAL_Buttons.c — interruption sur les deux flancs de PJ0/PJ1 (exercice 1.2). */
#include <stdbool.h>
#include <stdint.h>

#include "inc/hw_memmap.h"
#include "driverlib/gpio.h"
#include "driverlib/interrupt.h"
#include "driverlib/sysctl.h"

#include "HAL/HAL_Buttons.h"

/* Partagées entre l'ISR et main : volatile obligatoire. */
static volatile ButtonEvent_t s_sw1Event = eBUTTON_NO_EVENT;
static volatile ButtonEvent_t s_sw2Event = eBUTTON_NO_EVENT;

static void button_ISR(void)
{
    uint32_t const status = GPIOIntStatus(GPIO_PORTJ_BASE, true);

    if ((status & eBUTTON_USR_SW1) != 0u)
    {
        GPIOIntClear(GPIO_PORTJ_BASE, eBUTTON_USR_SW1);      /* acquitter TÔT */
        s_sw1Event = buttonState(eBUTTON_USR_SW1) ? eBUTTON_EVENT_PRESSED
                                                  : eBUTTON_EVENT_RELEASED;
    }
    else if ((status & eBUTTON_USR_SW2) != 0u)
    {
        GPIOIntClear(GPIO_PORTJ_BASE, eBUTTON_USR_SW2);
        s_sw2Event = buttonState(eBUTTON_USR_SW2) ? eBUTTON_EVENT_PRESSED
                                                  : eBUTTON_EVENT_RELEASED;
    }
}

void HAL_Button_Init(void)
{
    SysCtlPeripheralEnable(SYSCTL_PERIPH_GPIOJ);
    while (!SysCtlPeripheralReady(SYSCTL_PERIPH_GPIOJ))
    {
    }

    /* 1. la broche : entrée avec pull-up */
    GPIOPinTypeGPIOInput(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2));
    GPIOPadConfigSet(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2),
                     GPIO_STRENGTH_2MA, GPIO_PIN_TYPE_STD_WPU);

    /* 2. la source d'interruption : les deux flancs */
    GPIOIntTypeSet(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2),
                   GPIO_BOTH_EDGES);

    /* 3. le vecteur puis le NVIC, enfin le masque global */
    GPIOIntRegister(GPIO_PORTJ_BASE, button_ISR);
    GPIOIntEnable(GPIO_PORTJ_BASE, GPIO_INT_PIN_0 | GPIO_INT_PIN_1);
    IntMasterEnable();
}

bool buttonState(Button_t button)
{
    return (GPIOPinRead(GPIO_PORTJ_BASE, (uint8_t)button) == 0);
}

ButtonEvent_t HAL_Button_Event(Button_t button)
{
    ButtonEvent_t event;

    /* Lire puis effacer n'est pas atomique : l'ISR peut se glisser entre les deux. */
    IntMasterDisable();

    if (button == eBUTTON_USR_SW1)
    {
        event = s_sw1Event;
        s_sw1Event = eBUTTON_NO_EVENT;
    }
    else
    {
        event = s_sw2Event;
        s_sw2Event = eBUTTON_NO_EVENT;
    }

    IntMasterEnable();
    return event;
}

/* HAL_Buttons.c — l'ISR dépose dans la FIFO, main la vide (exercice 1.3). */
#include <stdbool.h>
#include <stdint.h>

#include "inc/hw_memmap.h"
#include "driverlib/gpio.h"
#include "driverlib/interrupt.h"
#include "driverlib/sysctl.h"

#include "HAL/HAL_Buttons.h"
#include "Serial/SerialBuffer.h"

bool buttonState(Button_t button);

static void button_ISR(void)
{
    uint32_t const status = GPIOIntStatus(GPIO_PORTJ_BASE, true);

    /* L'ISR fait le minimum : acquitter, déposer un événement, sortir. */
    if ((status & eBUTTON_USR_SW1) != 0u)
    {
        GPIOIntClear(GPIO_PORTJ_BASE, eBUTTON_USR_SW1);
        if (!SerialBuffer_isFull())
        {
            (void)SerialBuffer_addData((char)(buttonState(eBUTTON_USR_SW1)
                                              ? eBUTTON_EVENT_SW1_PRESSED
                                              : eBUTTON_EVENT_SW1_RELEASED));
        }
    }
    else if ((status & eBUTTON_USR_SW2) != 0u)
    {
        GPIOIntClear(GPIO_PORTJ_BASE, eBUTTON_USR_SW2);
        if (!SerialBuffer_isFull())
        {
            (void)SerialBuffer_addData((char)(buttonState(eBUTTON_USR_SW2)
                                              ? eBUTTON_EVENT_SW2_PRESSED
                                              : eBUTTON_EVENT_SW2_RELEASED));
        }
    }
}

void HAL_Button_Init(void)
{
    SysCtlPeripheralEnable(SYSCTL_PERIPH_GPIOJ);
    while (!SysCtlPeripheralReady(SYSCTL_PERIPH_GPIOJ))
    {
    }

    GPIOPinTypeGPIOInput(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2));
    GPIOPadConfigSet(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2),
                     GPIO_STRENGTH_2MA, GPIO_PIN_TYPE_STD_WPU);

    SerialBuffer_init();               /* la FIFO appartient à la HAL */

    GPIOIntTypeSet(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2),
                   GPIO_BOTH_EDGES);
    GPIOIntRegister(GPIO_PORTJ_BASE, button_ISR);
    GPIOIntEnable(GPIO_PORTJ_BASE, GPIO_INT_PIN_0 | GPIO_INT_PIN_1);
    IntMasterEnable();
}

bool buttonState(Button_t button)
{
    return (GPIOPinRead(GPIO_PORTJ_BASE, (uint8_t)button) == 0);
}

ButtonEvent_t HAL_Button_Event(void)
{
    char raw = 0;

    if (!SerialBuffer_getData(&raw))
    {
        return eBUTTON_NO_EVENT;
    }

    return (ButtonEvent_t)raw;
}

/* SerialBuffer.c — tampon circulaire de 16 octets (exercice 2.1, partie 1). */
#include <stdbool.h>
#include <stdint.h>

#include "Serial/SerialBuffer.h"

static char s_data[SERIAL_BUFFER_SIZE];
static uint16_t s_readIndex = 0u;
static uint16_t s_writeIndex = 0u;
static uint16_t s_lostData = 0u;

void SerialBuffer_init(void)
{
    s_readIndex = 0u;
    s_writeIndex = 0u;
    s_lostData = 0u;
}

bool SerialBuffer_isEmpty(void)
{
    return (s_readIndex == s_writeIndex);
}

bool SerialBuffer_isFull(void)
{
    /* Une case reste libre : c'est elle qui distingue « plein » de « vide ». */
    return ((uint16_t)((s_writeIndex + 1u) % SERIAL_BUFFER_SIZE) == s_readIndex);
}

bool SerialBuffer_addData(char value)
{
    if (SerialBuffer_isFull())
    {
        s_lostData++;                    /* FIFO pleine : on perd, on ne bloque pas */
        return false;
    }

    s_data[s_writeIndex] = value;         /* écrire d'abord... */
    s_writeIndex = (uint16_t)((s_writeIndex + 1u) % SERIAL_BUFFER_SIZE);   /* ...puis avancer */
    return true;
}

bool SerialBuffer_getData(char *value)
{
    if (SerialBuffer_isEmpty())
    {
        return false;
    }

    *value = s_data[s_readIndex];         /* lire d'abord... */
    s_readIndex = (uint16_t)((s_readIndex + 1u) % SERIAL_BUFFER_SIZE);     /* ...puis avancer */
    return true;
}

/* main.c — exercice 1.4 : la FIFO appartient à l'application, la HAL l'alimente. */
#include <stdbool.h>
#include <stdint.h>

#include "HAL/HAL_TM4C1294.h"
#include "HAL/HAL_Buttons.h"
#include "HAL/HAL_Leds.h"
#include "Serial/SerialBuffer.h"

static SerialBuffer_t g_events;      /* la file est ici, pas dans la HAL */

/* Appelée DEPUIS L'ISR : elle doit rester courte. */
static void onButtonEvent(Button_t button, ButtonEvent_t event)
{
    (void)button;                    /* l'événement encode déjà le bouton */

    if (!g_events.isFull(&g_events))
    {
        (void)g_events.addData(&g_events, (char)event);
    }
}

static void traiterEvenement(ButtonEvent_t event)
{
    switch (event)
    {
        case eBUTTON_EVENT_SW1_PRESSED:  HAL_Led_Write(eLED1, true);  break;
        case eBUTTON_EVENT_SW1_RELEASED: HAL_Led_Write(eLED1, false); break;
        case eBUTTON_EVENT_SW2_PRESSED:  HAL_Led_Write(eLED2, true);  break;
        case eBUTTON_EVENT_SW2_RELEASED: HAL_Led_Write(eLED2, false); break;
        default: break;
    }
}

int main(void)
{
    char raw;

    clockInit();
    HAL_Button_Init();
    HAL_Led_Init();

    SerialBuffer_init(&g_events);
    HAL_Button_RegisterCallback(onButtonEvent);

    while (1)
    {
        while (g_events.getData(&g_events, &raw))
        {
            traiterEvenement((ButtonEvent_t)raw);
        }
    }
}

/* HAL_Buttons.c — l'ISR appelle le callback, sans connaître l'application (exercice 1.4). */
#include <stdbool.h>
#include <stddef.h>      /* NULL */
#include <stdint.h>

#include "inc/hw_memmap.h"
#include "driverlib/gpio.h"
#include "driverlib/interrupt.h"
#include "driverlib/sysctl.h"

#include "HAL/HAL_Buttons.h"

/* static du MODULE : le pointeur doit survivre à la fonction d'enregistrement. */
static ButtonCallback_t s_callback = NULL;

static void button_ISR(void)
{
    uint32_t const status = GPIOIntStatus(GPIO_PORTJ_BASE, true);

    if ((status & eBUTTON_USR_SW1) != 0u)
    {
        GPIOIntClear(GPIO_PORTJ_BASE, eBUTTON_USR_SW1);
        if (s_callback != NULL)
        {
            s_callback(eBUTTON_USR_SW1, buttonState(eBUTTON_USR_SW1)
                                        ? eBUTTON_EVENT_SW1_PRESSED
                                        : eBUTTON_EVENT_SW1_RELEASED);
        }
    }
    else if ((status & eBUTTON_USR_SW2) != 0u)
    {
        GPIOIntClear(GPIO_PORTJ_BASE, eBUTTON_USR_SW2);
        if (s_callback != NULL)
        {
            s_callback(eBUTTON_USR_SW2, buttonState(eBUTTON_USR_SW2)
                                        ? eBUTTON_EVENT_SW2_PRESSED
                                        : eBUTTON_EVENT_SW2_RELEASED);
        }
    }
}

void HAL_Button_RegisterCallback(ButtonCallback_t callback)
{
    s_callback = callback;
}

void HAL_Button_Init(void)
{
    SysCtlPeripheralEnable(SYSCTL_PERIPH_GPIOJ);
    while (!SysCtlPeripheralReady(SYSCTL_PERIPH_GPIOJ))
    {
    }

    GPIOPinTypeGPIOInput(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2));
    GPIOPadConfigSet(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2),
                     GPIO_STRENGTH_2MA, GPIO_PIN_TYPE_STD_WPU);

    GPIOIntTypeSet(GPIO_PORTJ_BASE, (uint8_t)(eBUTTON_USR_SW1 | eBUTTON_USR_SW2),
                   GPIO_BOTH_EDGES);
    GPIOIntRegister(GPIO_PORTJ_BASE, button_ISR);
    GPIOIntEnable(GPIO_PORTJ_BASE, GPIO_INT_PIN_0 | GPIO_INT_PIN_1);
    IntMasterEnable();
}

bool buttonState(Button_t button)
{
    return (GPIOPinRead(GPIO_PORTJ_BASE, (uint8_t)button) == 0);
}

/* SerialBuffer.c — tampon circulaire générique (exercice 2.1, partie 2). */
#include <stdbool.h>
#include <stdint.h>

#include "Serial/SerialBuffer.h"

static bool prv_isEmpty(SerialBuffer_t *self)
{
    return (self->readIndex == self->writeIndex);
}

static bool prv_isFull(SerialBuffer_t *self)
{
    return ((uint16_t)((self->writeIndex + 1u) % SERIAL_BUFFER_SIZE) == self->readIndex);
}

static bool prv_addData(SerialBuffer_t *self, char value)
{
    if (self->isFull(self))
    {
        self->lostData++;
        return false;
    }

    self->data[self->writeIndex] = value;
    self->writeIndex = (uint16_t)((self->writeIndex + 1u) % SERIAL_BUFFER_SIZE);
    return true;
}

static bool prv_getData(SerialBuffer_t *self, char *value)
{
    if (self->isEmpty(self))
    {
        return false;
    }

    *value = self->data[self->readIndex];
    self->readIndex = (uint16_t)((self->readIndex + 1u) % SERIAL_BUFFER_SIZE);
    return true;
}

void SerialBuffer_init(SerialBuffer_t *self)
{
    self->readIndex = 0u;
    self->writeIndex = 0u;
    self->lostData = 0u;

    /* Les quatre méthodes sont choisies ici, une fois pour toutes. */
    self->addData = prv_addData;
    self->getData = prv_getData;
    self->isEmpty = prv_isEmpty;
    self->isFull = prv_isFull;
}
